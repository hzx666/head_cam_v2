cmake_minimum_required(VERSION 3.16)

project(head_cam_upload VERSION 1.0.0 LANGUAGES C ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

option(SERVICE_TESTING "Enable offline test-only dependency overrides" OFF)
option(SERVICE_STATIC "Statically link the service and dependencies" OFF)
set(BUNDLE_DIR "" CACHE PATH "Private output directory from embed_bundle.py")
set(SERVICE_DEPS "" CACHE PATH "Extracted target development libraries")
find_package(Threads REQUIRED)
add_executable(head-cam-service
    src/unified_main.c src/service_lifecycle.c src/service_supervisor.c src/monitor_main.c src/monitor_scan.c src/monitor_mqtt.c
    src/service_main.c src/service_util.c src/service_bundle.c
    src/service_upload.c src/service_tasks.c src/service_http.c
    src/head_mcap.c src/tf_scan.c third_party/cjson/cJSON.c)
target_include_directories(head-cam-service PRIVATE include third_party)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/lifecycle_data.h"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_lifecycle.py" "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lifecycle_data.h"
    DEPENDS scripts/embed_lifecycle.py scripts/boot_autostart.sh scripts/manual_dhcp.sh config/monitor.json
    VERBATIM)
target_sources(head-cam-service PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/lifecycle_data.h")
target_include_directories(head-cam-service PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(head-cam-service PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64 CJSON_NESTING_LIMIT=64)
target_compile_options(head-cam-service PRIVATE -Wall -Wextra -Wpedantic -Wshadow -Wformat=2 -Werror=return-type)
if(SERVICE_TESTING)
    target_compile_definitions(head-cam-service PRIVATE SERVICE_TESTING)
endif()
if(BUNDLE_DIR)
    target_sources(head-cam-service PRIVATE "${BUNDLE_DIR}/bundle_data.S")
    target_include_directories(head-cam-service PRIVATE "${BUNDLE_DIR}")
    target_compile_definitions(head-cam-service PRIVATE SERVICE_EMBEDDED)
elseif(NOT SERVICE_TESTING)
    message(FATAL_ERROR "Release needs BUNDLE_DIR; use build_release.sh. Offline development: -DSERVICE_TESTING=ON")
endif()
if(SERVICE_DEPS)
    target_include_directories(head-cam-service BEFORE PRIVATE "${SERVICE_DEPS}/usr/include" "${SERVICE_DEPS}/usr/include/aarch64-linux-gnu")
    set(DEP_LIB "${SERVICE_DEPS}/usr/lib/aarch64-linux-gnu")
    target_link_libraries(head-cam-service PRIVATE "${DEP_LIB}/libcrypto.a" "${DEP_LIB}/libzstd.a" "${DEP_LIB}/liblz4.a")
else()
    if(SERVICE_STATIC)
        set(CMAKE_FIND_LIBRARY_SUFFIXES .a)
    endif()
    find_library(CRYPTO_LIB crypto REQUIRED)
    find_library(ZSTD_LIB zstd REQUIRED)
    find_library(LZ4_LIB lz4 REQUIRED)
    target_link_libraries(head-cam-service PRIVATE ${CRYPTO_LIB} ${ZSTD_LIB} ${LZ4_LIB})
endif()
target_link_libraries(head-cam-service PRIVATE Threads::Threads m dl)

file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/SERVICE_VERSION" HEAD_CAM_RELEASE LIMIT_COUNT 1)
string(SUBSTRING "${HEAD_CAM_RELEASE}" 0 8 HEAD_CAM_SCHEMA)
target_compile_definitions(head-cam-service PRIVATE MONITOR_SCHEMA_VERSION=${HEAD_CAM_SCHEMA} MONITOR_RELEASE_VERSION="${HEAD_CAM_RELEASE}")
set_source_files_properties(src/service_main.c PROPERTIES COMPILE_DEFINITIONS main=upload_main)
set_source_files_properties(src/monitor_main.c PROPERTIES COMPILE_DEFINITIONS main=monitor_main)
set_source_files_properties(src/service_supervisor.c PROPERTIES COMPILE_DEFINITIONS main=supervisor_main)
set(BUSYBOX_BIN "" CACHE FILEPATH "Static target BusyBox embedded into unified binary")
if(BUSYBOX_BIN)
  file(SHA256 "${BUSYBOX_BIN}" BB_SHA)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/busybox_data.S" ".section .rodata\n.global embedded_busybox_start\n.global embedded_busybox_end\nembedded_busybox_start:\n.incbin \"${BUSYBOX_BIN}\"\nembedded_busybox_end:\n.section .note.GNU-stack,\"\",%progbits\n")
  target_sources(head-cam-service PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/busybox_data.S")
  target_compile_definitions(head-cam-service PRIVATE SERVICE_BUSYBOX BB_CACHE_TAG="${BB_SHA}")
elseif(NOT SERVICE_TESTING)
  message(FATAL_ERROR "Release requires BUSYBOX_BIN")
endif()
if(SERVICE_STATIC)
  target_link_options(head-cam-service PRIVATE -static)
endif()
add_custom_command(TARGET head-cam-service POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E create_symlink head-cam-service "${CMAKE_CURRENT_BINARY_DIR}/head-cam-upload"
  COMMAND ${CMAKE_COMMAND} -E create_symlink head-cam-service "${CMAKE_CURRENT_BINARY_DIR}/head-cam-monitor")
install(TARGETS head-cam-service RUNTIME DESTINATION bin)
